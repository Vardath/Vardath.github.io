<script>
document.addEventListener('DOMContentLoaded',()=>{
  const mobile=()=>window.matchMedia('(max-width:900px)').matches;
  const bar=document.querySelector('.bar');
  const toc=document.getElementById('toc');
  const toggle=document.querySelector('.mobile-nav-toggle');
  if(!bar||!toc||!toggle)return;

  const syncHeaderHeight=()=>{
    if(!mobile())return;
    const h=Math.max(58,Math.ceil(bar.getBoundingClientRect().height));
    document.documentElement.style.setProperty('--mobile-cosmology-header-height',h+'px');
  };

  const oldBackdrop=document.querySelector('.mobile-cosmology-nav-backdrop');
  if(oldBackdrop)oldBackdrop.remove();

  let grip=toc.querySelector('.mobile-cosmology-nav-grip');
  if(!grip){
    grip=document.createElement('div');
    grip.className='mobile-cosmology-nav-grip';
    grip.setAttribute('role','separator');
    grip.setAttribute('aria-label','Drag to resize section navigation width');
    grip.setAttribute('aria-orientation','vertical');
    toc.appendChild(grip);
  }else{
    grip.setAttribute('aria-label','Drag to resize section navigation width');
    grip.setAttribute('aria-orientation','vertical');
  }

  const sync=open=>{
    toc.classList.toggle('open',open);
    toggle.setAttribute('aria-expanded',String(open));
    toggle.textContent=open&&mobile()?'✕ Close':'☰ Sections';
    syncHeaderHeight();
  };
  const close=()=>sync(false);

  toggle.addEventListener('click',e=>{
    if(!mobile())return;
    e.preventDefault();
    e.stopImmediatePropagation();
    sync(!toc.classList.contains('open'));
  },true);

  toc.addEventListener('click',e=>{
    if(mobile()&&e.target.closest('a'))close();
  });
  document.addEventListener('keydown',e=>{
    if(e.key==='Escape'&&toc.classList.contains('open'))close();
  });

  let dragging=false;
  const resizeFromPointer=e=>{
    if(!dragging||!mobile())return;
    const min=230;
    const max=Math.max(min,Math.min(430,window.innerWidth-48));
    const width=Math.max(min,Math.min(max,window.innerWidth-e.clientX-6));
    document.documentElement.style.setProperty('--mobile-cosmology-drawer-width',width+'px');
  };
  grip.addEventListener('pointerdown',e=>{
    if(!mobile())return;
    dragging=true;
    grip.setPointerCapture?.(e.pointerId);
    e.preventDefault();
  });
  grip.addEventListener('pointermove',resizeFromPointer);
  const stopDrag=e=>{
    if(!dragging)return;
    dragging=false;
    try{grip.releasePointerCapture?.(e.pointerId)}catch(_){ }
  };
  grip.addEventListener('pointerup',stopDrag);
  grip.addEventListener('pointercancel',stopDrag);

  const refresh=()=>{
    syncHeaderHeight();
    if(!mobile()){
      toc.classList.remove('open');
      toggle.setAttribute('aria-expanded','false');
      toggle.textContent='☰ Sections';
    }
  };

  syncHeaderHeight();
  requestAnimationFrame(syncHeaderHeight);
  setTimeout(syncHeaderHeight,120);
  window.addEventListener('resize',refresh);
  window.addEventListener('orientationchange',()=>setTimeout(syncHeaderHeight,100));
  if('ResizeObserver' in window)new ResizeObserver(syncHeaderHeight).observe(bar);
});
</script>
