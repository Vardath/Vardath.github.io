<script>
document.addEventListener('DOMContentLoaded',()=>{
  const mobile=()=>window.matchMedia('(max-width:900px)').matches;
  const bar=document.querySelector('.bar');
  const toc=document.getElementById('toc');
  const toggle=document.querySelector('.mobile-nav-toggle');
  if(!bar||!toc)return;

  const cleanupOldDrawer=()=>{
    document.querySelectorAll('.mobile-cosmology-nav-grip,.mobile-cosmology-nav-backdrop').forEach(el=>el.remove());
    toc.classList.remove('open');
    if(toggle){
      toggle.setAttribute('aria-expanded','false');
      toggle.textContent='☰ Sections';
    }
  };

  const syncHeaderHeight=()=>{
    if(!mobile())return;
    const h=Math.max(68,Math.ceil(bar.getBoundingClientRect().height));
    document.documentElement.style.setProperty('--mobile-cosmology-header-height',h+'px');
  };

  cleanupOldDrawer();
  syncHeaderHeight();
  requestAnimationFrame(syncHeaderHeight);
  setTimeout(syncHeaderHeight,120);

  const keepActiveVisible=()=>{
    if(!mobile())return;
    const active=toc.querySelector('a.active');
    if(active)active.scrollIntoView({block:'nearest',inline:'nearest'});
  };

  const observer=new MutationObserver(keepActiveVisible);
  observer.observe(toc,{subtree:true,attributes:true,attributeFilter:['class']});

  const refresh=()=>{
    cleanupOldDrawer();
    syncHeaderHeight();
    keepActiveVisible();
  };

  window.addEventListener('resize',refresh);
  window.addEventListener('orientationchange',()=>setTimeout(refresh,100));
  if('ResizeObserver' in window)new ResizeObserver(syncHeaderHeight).observe(bar);
});
</script>
