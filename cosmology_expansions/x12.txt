<script>
document.addEventListener('DOMContentLoaded',()=>{
  const mobile=()=>window.matchMedia('(max-width:900px)').matches;
  const bar=document.querySelector('.bar');
  const toc=document.getElementById('toc');
  const toggle=document.querySelector('.mobile-nav-toggle');
  if(!bar||!toc||!toggle)return;

  let backdrop=document.querySelector('.mobile-cosmology-nav-backdrop');
  if(!backdrop){
    backdrop=document.createElement('div');
    backdrop.className='mobile-cosmology-nav-backdrop';
    backdrop.setAttribute('aria-hidden','true');
    document.body.appendChild(backdrop);
  }

  let grip=toc.querySelector('.mobile-cosmology-nav-grip');
  if(!grip){
    grip=document.createElement('div');
    grip.className='mobile-cosmology-nav-grip';
    grip.setAttribute('role','separator');
    grip.setAttribute('aria-label','Drag to resize section navigation');
    grip.setAttribute('aria-orientation','horizontal');
    toc.appendChild(grip);
  }

  const sync=open=>{
    toc.classList.toggle('open',open);
    backdrop.classList.toggle('open',open&&mobile());
    document.body.classList.toggle('mobile-cosmology-nav-open',open&&mobile());
    toggle.setAttribute('aria-expanded',String(open));
    toggle.textContent=open&&mobile()?'✕ Close':'☰ Sections';
  };
  const close=()=>sync(false);

  // Capture phase prevents the original mobile toggle listener from toggling a second time.
  toggle.addEventListener('click',e=>{
    if(!mobile())return;
    e.preventDefault();
    e.stopImmediatePropagation();
    sync(!toc.classList.contains('open'));
  },true);

  backdrop.addEventListener('click',close);
  toc.addEventListener('click',e=>{
    if(mobile()&&e.target.closest('a'))close();
  });
  document.addEventListener('keydown',e=>{
    if(e.key==='Escape'&&toc.classList.contains('open'))close();
  });

  let dragging=false;
  const resizeFromPointer=e=>{
    if(!dragging||!mobile())return;
    const top=toc.getBoundingClientRect().top;
    const min=220;
    const max=Math.max(min,window.innerHeight-top-12);
    const h=Math.max(min,Math.min(max,e.clientY-top));
    document.documentElement.style.setProperty('--mobile-cosmology-nav-height',h+'px');
  };
  grip.addEventListener('pointerdown',e=>{
    if(!mobile())return;
    dragging=true;
    grip.setPointerCapture?.(e.pointerId);
    e.preventDefault();
  });
  grip.addEventListener('pointermove',resizeFromPointer);
  const stopDrag=e=>{
    if(!dragging)return;
    dragging=false;
    try{grip.releasePointerCapture?.(e.pointerId)}catch(_){ }
  };
  grip.addEventListener('pointerup',stopDrag);
  grip.addEventListener('pointercancel',stopDrag);

  window.addEventListener('resize',()=>{
    if(!mobile()){
      backdrop.classList.remove('open');
      document.body.classList.remove('mobile-cosmology-nav-open');
      toggle.textContent='☰ Sections';
    }else if(toc.classList.contains('open')){
      backdrop.classList.add('open');
      document.body.classList.add('mobile-cosmology-nav-open');
      toggle.textContent='✕ Close';
    }
  });
});
</script>
