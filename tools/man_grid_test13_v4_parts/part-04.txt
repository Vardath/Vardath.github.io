def run(shard):
    if not 0 <= shard < N:
        raise SystemExit('shard must be 0..19')
    model = read_gz(WORK / 'model.json.gz')
    validate_model(model)
    costs = substitution_costs(model)
    rows = read_gz(WORK / f'historical-shard-{shard:02d}.json.gz')
    out = []
    lineage_rows = defaultdict(list)
    transition = Counter(); op_counts = Counter()
    move_sums = [0.0] * 20; move_n = 0

    for r in rows:
        anc = medoid(r['ancestor_forms'], costs)
        des = medoid(r['descendant_forms'], costs)
        if not anc or not des:
            continue
        real = seqdist(anc, des, costs)
        aln = alignment(anc, des, costs)
        mv, events = movement_from_alignment(aln, model)
        for e in events:
            op_counts[e['op']] += 1
            if e.get('a') is not None and e.get('b') is not None:
                transition[(e['a'], e['b'])] += 1
        if mv is not None:
            move_n += 1
